95def canonical_subtype(raw_value: str) -> str: 

96 """Normalize user input to a known concept subtype when possible. 

97 

98 Unknown values are slug-ified and returned as-is so the system remains 

99 open to any cuisine type rather than rejecting unfamiliar inputs. 

100 """ 

101 normalized = raw_value.strip().lower().replace("-", "_").replace(" ", "_") 

102 if normalized in HEALTHY_SUBTYPES: 

103 return normalized 

104 # Try keyword matching in the raw (lowered) text 

105 lowered = raw_value.strip().lower() 

106 for subtype, keywords in _KEYWORDS.items(): 

107 if any(kw in lowered for kw in keywords): 

108 return subtype 

109 # Return slug of whatever the user typed — custom subtype 

110 return normalized
